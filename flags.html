regionNames = new Intl.DisplayNames(["en"], { type: "region" })

subdivisionNames = new Map([
  ["gb-eng", "England"],
  ["gb-nir", "Northern Ireland"],
  ["gb-sct", "Scotland"],
  ["gb-wls", "Wales"]
])

flagOptions = flagCodes.map(code => ({
  code,
  name: subdivisionNames.get(code) || regionNames.of(code.toUpperCase()) || code.toUpperCase()
}))

viewof selectedFlag = Inputs.select(flagOptions, {
  label: "Select a flag",
  format: flag => flag.name
})

html`<figure>
  <img
    src="flags/${selectedFlag.code}.png"
    alt="Flag of ${selectedFlag.name}"
    style="display: block; width: min(100%, 640px); height: auto;"
  >
  <figcaption>${selectedFlag.name}</figcaption>
</figure>`
